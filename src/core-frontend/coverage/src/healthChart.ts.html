export interface Segment { from: number; to: number; state: string }
 
/** Runs of the same bot state over time: each segment lasts until the next sample (the last one until `end`). */
export function stateSegments(samples: { ts: number; botState: string | null }[], end: number): Segment[] {
  const out: Segment[] = [];
  samples.forEach((s, i) => {
    const state = s.botState ?? 'unknown';
    const to = i + 1 < samples.length ? samples[i + 1].ts : Math.max(end, s.ts);
    const prev = out[out.length - 1];
    if (prev && prev.state === state) prev.to = to;
    else out.push({ from: s.ts, to, state });
  });
  return out;
}
 
/** Share of the time (0..1) the samples spent in a state for which `good` holds. */
export function shareOf(segs: Segment[], good: (state: string) => boolean): number {
  const total = segs.reduce((n, s) => n + (s.to - s.from), 0);
  if (!total) return 1;
  return segs.filter((s) => good(s.state)).reduce((n, s) => n + (s.to - s.from), 0) / total;
}
 
export interface Pt { ts: number; v: number | null }
 
/** Keeps at most `max` points: neighbours are averaged, the time of a bucket is its middle. Gaps stay gaps. */
export function thin(points: Pt[], max: number): Pt[] {
  if (points.length <= max || max < 2) return points;
  const step = points.length / max;
  const out: Pt[] = [];
  for (let i = 0; i < max; i++) {
    const a = Math.floor(i * step);
    const b = Math.max(a + 1, Math.floor((i + 1) * step));
    const part = points.slice(a, b);
    const vals = part.map((p) => p.v).filter((v): v is number => v != null && Number.isFinite(v));
    out.push({ ts: part[Math.floor(part.length / 2)].ts, v: vals.length ? vals.reduce((s, v) => s + v, 0) / vals.length : null });
  }
  return out;
}
 
/** The index of the point nearest to a horizontal position given as a share of the width (0..1). */
export function nearestIndex(share: number, n: number): number {
  if (n <= 1) return 0;
  return Math.max(0, Math.min(n - 1, Math.round(share * (n - 1))));
}
