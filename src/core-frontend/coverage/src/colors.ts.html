export function toHsl([r, g, b]: RGB): [number, number, number] {
  const R = r / 255, G = g / 255, B = b / 255;
  const max = Math.max(R, G, B), min = Math.min(R, G, B);
  const l = (max + min) / 2;
  const d = max - min;
  if (d === 0) return [0, 0, l];
  const s = d / (1 - Math.abs(2 * l - 1));
  let h = max === R ? ((G - B) / d) % 6 : max === G ? (B - R) / d + 2 : (R - G) / d + 4;
  h = (h * 60 + 360) % 360;
  return [h, s, l];
}
 
export function fromHsl([h, s, l]: [number, number, number]): RGB {
  const c = (1 - Math.abs(2 * l - 1)) * s;
  const x = c * (1 - Math.abs(((h / 60) % 2) - 1));
  const m = l - c / 2;
  const [r, g, b] = h < 60 ? [c, x, 0] : h < 120 ? [x, c, 0] : h < 180 ? [0, c, x] : h < 240 ? [0, x, c] : h < 300 ? [x, 0, c] : [c, 0, x];
  return [r, g, b].map((v) => Math.round((v + m) * 255)) as RGB;
}
 
/**
 * Moves the lightness of `c` (hue and saturation kept) in the given direction until it has `min` contrast
 * with every colour in `against`. Returns the original colour when it already passes.
 */
export function ensureContrast(c: RGB, against: RGB[], min: number, direction: 'darken' | 'lighten'): RGB {
  const ok = (x: RGB) => against.every((bg) => contrast(x, bg) >= min);
  if (ok(c)) return c;
  const [h, s, l] = toHsl(c);
  const step = direction === 'darken' ? -0.01 : 0.01;
  let cur = c;
  for (let L = l + step; L >= 0 && L <= 1; L += step) {
    cur = fromHsl([h, s, clamp(L, 0, 1)]);
    if (ok(cur)) return cur;
  }
  return cur;
}

/**
 * CSS-variable values of the brand colour(s): `fill` (buttons, white text >= 4.5:1), `hover`, `end` (the other end
 * of the gradients: the second colour darkened until white text reads >= 4.5:1, or a darker mix of `fill`) and `link`
 * (brand-coloured text, >= 4.5:1 on the card and soft surfaces).
 */
export function brandVars(color: string, color2: string | undefined, dark: boolean, card: RGB, soft: RGB) {
  const base = parseHex(color) ?? ([79, 70, 229] as RGB);
  const fill = contrast(base, WHITE) >= 4.5 ? base : ensureContrast(base, [WHITE], 4.5, 'darken');
  const hover = mix(fill, BLACK, 0.14);
  const link = dark ? ensureContrast(mix(base, WHITE, 0.45), [card, soft], 4.5, 'lighten') : ensureContrast(base, [card, soft], 4.5, 'darken');
  const second = color2 ? parseHex(color2) : null;
  const end = second ? (contrast(second, WHITE) >= 4.5 ? second : ensureContrast(second, [WHITE], 4.5, 'darken')) : mix(fill, BLACK, 0.24);
  return { fill, hover, end, link };
}
 
/**
 * Two-stop gradient for a tile cover from a game colour: c1 is the deeper end (white text keeps >= 4.5:1),
 * c2 a lighter, slightly warmer end (large white text keeps >= 3:1).
 */
export function gradientFor(color: string): [string, string] {
  const base = parseHex(color) ?? [90, 75, 209];
  const c1 = ensureContrast(mix(base, BLACK, 0.1), [WHITE], 4.5, 'darken');
  const [h, s, l] = toHsl(mix(base, WHITE, 0.22));
  const warm = fromHsl([(h + 14) % 360, clamp(s, 0, 0.95), l]);
  const c2 = ensureContrast(warm, [WHITE], 3, 'darken');
  return [hex(c1), hex(c2)];
}
