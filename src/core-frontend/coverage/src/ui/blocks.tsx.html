/**
 * Dialog: a bottom sheet on phones, a centred card on larger screens.
 * Focus moves inside and is trapped, Escape closes, focus returns to the opener, only a real click on the backdrop closes it.
 */
export function Modal({ title, onClose, children, side = false }: { title: string; onClose: () => void; children: ReactNode; /** a drawer from the right edge (full height) instead of a centred dialog: the list stays visible behind it */ side?: boolean }) {
  // keyboards: first field; touch screens: the dialog itself (no on-screen keyboard popping up)
  const focusFirst = (e: Event) => {
    e.preventDefault();
    const box = (e.currentTarget as HTMLElement | null);
    if (!box) return;
    const target = window.matchMedia?.('(pointer: coarse)').matches
      ? box
      : box.querySelector<HTMLElement>('input:not([type="hidden"]):not([disabled]), select:not([disabled]), textarea:not([disabled])') ?? box;
    target.focus();
  };
  // phone: drag the grip (or the title) down to close the sheet
  const drag = useRef<{ y: number; el: HTMLElement } | null>(null);
  const onGripStart = (e: ReactTouch<HTMLElement>) => { const box = e.currentTarget.closest<HTMLElement>('[role="dialog"]'); if (box && box.scrollTop <= 0) drag.current = { y: e.touches[0].clientY, el: box }; };
  const onGripMove = (e: ReactTouch<HTMLElement>) => {
    const d = drag.current; if (!d) return;
    const dy = Math.max(0, e.touches[0].clientY - d.y);
    d.el.style.transition = 'none';
    d.el.style.transform = `translateY(${dy}px)`;
  };
  const onGripEnd = (e: ReactTouch<HTMLElement>) => {
    const d = drag.current; drag.current = null; if (!d) return;
    const dy = Math.max(0, e.changedTouches[0].clientY - d.y);
    d.el.style.transition = 'transform .2s ease';
    if (dy > 90) { d.el.style.transform = 'translateY(100%)'; window.setTimeout(onClose, 160); } else d.el.style.transform = '';
  };
  return (
    <Dialog.Root open onOpenChange={(o) => { if (!o) onClose(); }}>
      <Dialog.Portal>
        <Dialog.Overlay className={side ? 'modal-ov fixed inset-0 z-50 flex justify-end bg-black/45 p-0' : 'modal-ov fixed inset-0 z-50 flex items-end justify-center bg-black/55 p-0 backdrop-blur-xs sm:items-center sm:p-4'}>
          <Dialog.Content
            aria-describedby={undefined}
            onOpenAutoFocus={focusFirst}
            className={side
              ? 'drawer-in panel panel-raised h-full w-full max-w-md overflow-auto overscroll-contain rounded-none p-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] outline-hidden sm:rounded-l-ui-xl sm:p-6'
              : 'pop panel panel-raised max-h-[92dvh] w-full max-w-lg overflow-auto overscroll-contain rounded-b-none p-5 pb-[max(1.25rem,env(safe-area-inset-bottom))] outline-hidden sm:rounded-b-ui-xl sm:p-6'}
          >
            {!side && <div className="sheet-grip" aria-hidden onTouchStart={onGripStart} onTouchMove={onGripMove} onTouchEnd={onGripEnd}><i /></div>}
            <div className="mb-4 flex items-center justify-between gap-3" onTouchStart={side ? undefined : onGripStart} onTouchMove={side ? undefined : onGripMove} onTouchEnd={side ? undefined : onGripEnd}>
              <Dialog.Title className="min-w-0 text-xl font-black">{title}</Dialog.Title>
              <button type="button" className="btn-icon" onClick={onClose} aria-label={t('common.close')}>
                <IconClose />
              </button>
            </div>
            {children}
          </Dialog.Content>
        </Dialog.Overlay>
      </Dialog.Portal>
    </Dialog.Root>
  );
}

/** `glass`: for use on top of a cover or a gradient (dark translucent capsule, white text, coloured dot) */
export function StatusPill({ online, known, lag, glass = false, maintenance = false }: { online: boolean; known: boolean; lag?: boolean; glass?: boolean; maintenance?: boolean }) {
  const kind = maintenance ? 'maint' : !known ? 'mute' : !online ? 'bad' : lag ? 'lag' : 'ok';
  // a change of state (not the first paint) plays a short swap animation on the badge
  const prev = useRef(kind);
  const changed = prev.current !== kind;
  useEffect(() => { prev.current = kind; }, [kind]);
  const swap = changed ? ' pill-swap' : '';
  // a window of maintenance explains the silence: neither «Офлайн» nor «Лаги»
  if (maintenance) return glass ? <span className={`pill-glass pill-glass-maint${swap}`}>Обслуживание</span> : <span className={`pill pill-maint${swap}`}>ОБСЛУЖИВАНИЕ</span>;
  if (glass) {
    const text = !known ? 'Ожидание' : !online ? 'Офлайн' : lag ? 'Лаги' : 'Онлайн';
    return <span className={`pill-glass pill-glass-${kind}${swap}`}>{text}</span>;
  }
  if (!known) return <span className={`pill pill-mute${swap}`}>ОЖИДАНИЕ</span>;
  if (!online) return <span className={`pill pill-bad${swap}`}>ОФЛАЙН</span>;
  return lag ? <span className={`pill pill-warn lag${swap}`}>ЛАГИ</span> : <span className={`pill pill-ok live${swap}`}>ОНЛАЙН</span>;
}
 
/**
 * `glass`: translucent white square, for use on top of a game-coloured cover.
 * `vtId`: a server id; the avatar of that server morphs between the list tile and the server page (View Transitions).
 */
export function GameAvatar({ game, size = 44, glass = false, vtId }: { game: string; size?: number; glass?: boolean; vtId?: number }) {
  const icon = gameIconPath(game);
  const [c1, c2] = gameGradient(game);
  return (
    <div
      data-vt-avatar={vtId}
      className="grid shrink-0 place-items-center font-black text-white"
      style={{
        width: size,
        height: size,
        borderRadius: Math.round(size * 0.3),
        // artwork sits on a deep tile of the game colour (as on the PNG card), a line pictogram on the colour itself
        background: glass
          ? 'rgba(255,255,255,0.24)'
          : GAME_IMAGE[game]
            ? `linear-gradient(135deg, color-mix(in srgb, ${c1} 55%, #000), color-mix(in srgb, ${c2} 25%, #000))`
            : `linear-gradient(135deg, ${c1}, ${c2})`,
        fontSize: size * 0.34,
        boxShadow: glass ? 'inset 0 0 0 1px rgba(255,255,255,0.25)' : `0 4px 12px -4px ${c1}99`,
      }}
      aria-hidden
    >
      {GAME_IMAGE[game] ? (
        <img src={GAME_IMAGE[game]} width={size * 0.74} height={size * 0.74} alt="" draggable={false} className="object-contain" />
      ) : icon ? (
        <svg width={size * 0.58} height={size * 0.58} viewBox="0 0 24 24" fill="none" stroke="#fff" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round">
          <path d={icon} />
        </svg>
      ) : (
        gameAbbr(game)
      )}
    </div>
  );
}

/**
 * Page title row: icon, title, subtitle and the page actions on the right (one primary action, the rest soft).
 * `hint`: one sentence under the title (what this page is and why); `help`: longer text behind a «Подробнее» disclosure.
 */
export function PageHead({ title, subtitle, icon, hint, help, children }: { title: string; subtitle?: string; icon?: ReactNode; hint?: ReactNode; help?: ReactNode; children?: ReactNode }) {
  const [open, setOpen] = useState(false);
  const helpId = useId();
  // a hint is read once: the cross hides it (remembered per page in this browser), the «i» button next to the actions brings it back
  const hintKey = `coregsm_hint_${title}`;
  // a page's hint is shown on the first visit only; later visits start with it folded (the «i» button brings it back)
  const seenKey = `coregsm_hintseen_${title}`;
  const [hidden, setHidden] = useState(() => { try { return localStorage.getItem(hintKey) === '1' || localStorage.getItem(seenKey) === '1'; } catch { return false; } });
  useEffect(() => { try { localStorage.setItem(seenKey, '1'); } catch { /* not remembered */ } }, [seenKey]);
  const setHint = (off: boolean) => { setHidden(off); try { if (off) localStorage.setItem(hintKey, '1'); else localStorage.removeItem(hintKey); } catch { /* not remembered */ } };
  const hasHint = !!(hint || help);
  return (
    <div className="space-y-3">
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div className="flex min-w-0 items-center gap-3.5">
          {icon && <span className="grid h-11 w-11 shrink-0 place-items-center rounded-ui-lg bg-linear-to-br from-brand to-brand-2 text-white shadow-pop sm:h-12 sm:w-12">{icon}</span>}
          <div className="min-w-0">
            <h1 className="text-xl font-extrabold tracking-tight sm:text-[30px]">{title}</h1>
            {subtitle && <p className="mt-0.5 hidden font-semibold text-mute sm:block">{subtitle}</p>}
          </div>
        </div>
        {(children || (hasHint && hidden)) && (
          <div className="flex flex-wrap items-center gap-2">
            {hasHint && hidden && <button type="button" className="btn-icon-ghost btn-icon-sm" title="Показать подсказку" aria-label="Показать подсказку" onClick={() => setHint(false)}><span aria-hidden className="hint-ico">i</span></button>}
            {children}
          </div>
        )}
      </div>
      {hasHint && !hidden && (
        <div className="hint-strip">
          <span aria-hidden className="hint-ico">i</span>
          <div className="min-w-0 flex-1">
            <p>
              {hint}
              {help && (
                <>
                  {hint ? ' ' : ''}
                  <button type="button" className="hint-more" aria-expanded={open} aria-controls={helpId} onClick={() => setOpen((v) => !v)}>
                    {open ? 'Скрыть' : 'Подробнее'}
                  </button>
                </>
              )}
            </p>
            {help && open && <div id={helpId} className="hint-help">{help}</div>}
          </div>
          <button type="button" className="hint-close" aria-label="Скрыть подсказку" title="Скрыть подсказку" onClick={() => setHint(true)}>✕</button>
        </div>
      )}
    </div>
  );
}

/** a coloured strip with a message and an optional action (usually a GoLink) */

/** Copies text to the clipboard; falls back to a dialog with the text selected where the Clipboard API is blocked. */
export function CopyButton({ text, label = 'Копировать', className = '', tone = 'soft' }: { text: string; label?: string; className?: string; tone?: 'soft' | 'grad' | 'glass' }) {
  const [done, setDone] = useState(false);
  const timer = useRef<number | undefined>(undefined);
  useEffect(() => () => window.clearTimeout(timer.current), []);
  const copy = async () => {
    try {
      await navigator.clipboard.writeText(text);
      setDone(true);
      window.clearTimeout(timer.current);
      timer.current = window.setTimeout(() => setDone(false), 1500);
    } catch {
      await promptDialog({ title: 'Скопируйте вручную', label: 'Текст', value: text, readOnly: true });
    }
  };
  return (
    <button type="button" className={`${tone === 'grad' ? 'btn-on-grad btn-sm' : tone === 'glass' ? 'btn-glass btn-sm' : 'btn btn-soft btn-sm'} ${className}`} onClick={copy} title={label}>
      <span className="copy-ico" data-done={done} aria-hidden>
        <span className="ico-copy inline-grid"><IconCopy /></span>
        <svg className="ico-check" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round">
          <path d="M5 12.5l4.5 4.5L19 7.500" />
        </svg>
        {done && <Burst />}
      </span>
      <span className="min-w-0 truncate" aria-live="polite">{done ? 'Скопировано' : label}</span>
    </button>
  );
}

  id,
  value,
  type,

  onType,
  placeholder,
  disabled,

  id: string;
  value: string;

          id={id}
          className="input font-mono"
          value={value}
          disabled={disabled}

      </p>
    </div>
  );
}

/** a few grey rows while a list loads: the page keeps its shape instead of jumping when the data comes */
export function SkeletonRows({ n = 3, h = 'h-14' }: { n?: number; h?: string }) {
  return (
    <div className="grid gap-2" aria-busy="true" role="status" aria-label="Загрузка">

  );
}
